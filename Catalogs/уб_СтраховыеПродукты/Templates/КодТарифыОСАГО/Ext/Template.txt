
мЯвляетсяДопсом	= ЕстьРеквизитДокумента("ИсходныйДоговор", ДокументОбъект.Метаданные());
КонецПериода	= ?(мЯвляетсяДопсом, ДокументОбъект.Дата, Неопределено);
Договор			= ?(мЯвляетсяДопсом, ДокументОбъект.ИсходныйДоговор, ДокументОбъект.Ссылка);
Дата    		= ДокументОбъект.Дата;

// получим шапку
ШапкаДСП		= СформироватьШапкуИзРегистра(Договор, "ШапкаДСП", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен, КонецПериода);
Страхователь	= ШапкаДСП.Контрагент;
ДатаНачала		= ШапкаДСП.ДатаНачала;
ДатаОкончания	= ШапкаДСП.ДатаОкончания;

// ТС
ТзОбъектыИРиски	= ПолучитьДвиженияДокументаПоРегиструСведений(Договор, "ОбъектыИРискиДСП");

Если ТзОбъектыИРиски.Количество() = 0 Тогда
	Предупреждение("Не указаны объекты и риски. Печать невозможна!");
Иначе

	ОбъектОСАГО		= ТзОбъектыИРиски[0].Объект;
	СтОтбораТС		= Новый Структура("Объект", ОбъектОСАГО);
	ТС				= РегистрыСведений.ТранспортныеСредства.Получить(СтОтбораТС);

	// получим ДанныеПоОбъектуОСАГО
	ДанныеПоОбъектуОСАГО	= СформироватьШапкуИзРегистраПоОбъекту(ОбъектОСАГО, "ДанныеПоОбъектуОСАГО", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен);
	Собственник				= ДанныеПоОбъектуОСАГО.Собственник;
	Версия					= ДанныеПоОбъектуОСАГО.Версия;
	ТипТСПоОСАГО			= ДанныеПоОбъектуОСАГО.ТипТСПоОСАГО;
	Если Версия=1 Тогда
		ДатаВерсии	= '20051231';
	Иначе
		ДатаВерсии	= Неопределено;
	КонецЕсли;

	Если	(ТипТСПоОСАГО=Перечисления.ТипыТСПоОСАГО.ЛегковойАвтомобильЮр)	ИЛИ
			(ТипТСПоОСАГО=Перечисления.ТипыТСПоОСАГО.ЛегковойАвтомобильФиз)	ИЛИ
			(ТипТСПоОСАГО=Перечисления.ТипыТСПоОСАГО.Такси) 				Тогда
		ЭтоЛегковой	= Истина;
	Иначе
		ЭтоЛегковой	= Ложь;
	КонецЕсли;
	Если	(ТипТСПоОСАГО=Перечисления.ТипыТСПоОСАГО.ПрицепЛегковогоАвтомобиля)	ИЛИ
			(ТипТСПоОСАГО=Перечисления.ТипыТСПоОСАГО.ПрицепГрузовогоАвтомобиля)	ИЛИ
			(ТипТСПоОСАГО=Перечисления.ТипыТСПоОСАГО.ПрицепТрактора)			Тогда
		ЭтоПрицеп	= Истина;
	Иначе
		ЭтоПрицеп	= Ложь;
	КонецЕсли;

	//	Премия,ТБ,КБМ,КВС,КО,КМ,КП,КН,КТ,КС,Формула,Класс,ВозрастСтаж
	Результат			= Новый Структура;
	ФормулаПремияДопс	= "";

	// водители и их класс
	ТзВодителиТСдляОСАГО = Новый ТаблицаЗначений;
	ДобавитьКолонкуВТЗ(ТзВодителиТСдляОСАГО, "Объект");
	ДобавитьКолонкуВТЗ(ТзВодителиТСдляОСАГО, "Контрагент");
	ДобавитьКолонкуВТЗ(ТзВодителиТСдляОСАГО, "Наличие");
	СформироватьНачТЗПоРегиструПоИзмерению("Объект", ОбъектОСАГО, ТзВодителиТСдляОСАГО, , "ВодителиТСдляОСАГО", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен, КонецПериода);

	ТзКБМпоВодителям = Новый ТаблицаЗначений;
	ДобавитьКолонкуВТЗ(ТзКБМпоВодителям, "Контрагент");
	ДобавитьКолонкуВТЗ(ТзКБМпоВодителям, "Класс");

	ВодителиТСсКБМ		= ТзКБМпоВодителям.Скопировать();

	ТекКБМпоВодителям	= ТзКБМпоВодителям.Скопировать();
	Для Каждого Водитель Из ТзВодителиТСдляОСАГО Цикл
		ТекКБМпоВодителям.Очистить();
		СформироватьНачТЗПоРегиструПоИзмерению("Контрагент", Водитель.Контрагент, ТекКБМпоВодителям, , "КБМпоВодителям", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен, КонецПериода);
		Если ТекКБМпоВодителям.Количество()>0 Тогда
			НовВодитель				= ТзКБМпоВодителям.Добавить();
			НовВодитель.Контрагент	= ТекКБМпоВодителям[0].Контрагент;
			НовВодитель.Класс		= ТекКБМпоВодителям[0].Класс;
		Конецесли;
	КонецЦикла;

	Для Каждого Водитель Из ТзВодителиТСдляОСАГО Цикл
		НовСтрВодителиТСсКБМ			= ВодителиТСсКБМ.Добавить();
		НовСтрВодителиТСсКБМ.Контрагент	= Водитель.Контрагент;
		
		СтрокаВодитель	= ТзКБМпоВодителям.Найти(Водитель.Контрагент, "Контрагент");
		Если СтрокаВодитель<>Неопределено Тогда
			НовСтрВодителиТСсКБМ.Класс	= СтрокаВодитель.Класс;
		КонецЕсли;
	КонецЦикла;

	// Период использования ТС
	ТзПериодыИспользованияТС = Новый ТаблицаЗначений;
	ДобавитьКолонкуВТЗ(ТзПериодыИспользованияТС, "Объект");
	ДобавитьКолонкуВТЗ(ТзПериодыИспользованияТС, "ДатаНачала");
	ДобавитьКолонкуВТЗ(ТзПериодыИспользованияТС, "ДатаОкончания");
	ДобавитьКолонкуВТЗ(ТзПериодыИспользованияТС, "Наличие");
	СформироватьНачТЗПоРегиструПоИзмерению("Объект", ОбъектОСАГО, ТзПериодыИспользованияТС, , "ПериодыИспользованияТС", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен, КонецПериода);

	ПериодыДляВвода	= Новый ТаблицаЗначений;
	ПериодыДляВвода.Колонки.Добавить("ДатаНачала");
	ПериодыДляВвода.Колонки.Добавить("ДатаОкончания");
	Для Каждого Стр Из ТзПериодыИспользованияТС Цикл
		НовСтрПериодыДляВвода	= ПериодыДляВвода.Добавить();
		НовСтрПериодыДляВвода.ДатаНачала	= Стр.ДатаНачала;
		НовСтрПериодыДляВвода.ДатаОкончания	= Стр.ДатаОкончания;
	КонецЦикла;

	СтДат			= Новый Структура("ДатаНачала, ДатаОкончания", ДатаНачала, ДатаОкончания);
	СтДанныхОСАГО	= Новый Структура;
	СтДанныхОСАГО.Вставить("Категория", ТС.Категория);
	СтДанныхОСАГО.Вставить("МестоРегистрации", ТС.МестоРегистрации);
	СтДанныхОСАГО.Вставить("МощностьДвигателя", ТС.МощностьДвигателя);
	СтДанныхОСАГО.Вставить("ТипТСПоОСАГО", ТипТСПоОСАГО);
	СтДанныхОСАГО.Вставить("Собственник", Собственник);
	СтДанныхОСАГО.Вставить("СписокЛицУправленияОграничен", ДанныеПоОбъектуОСАГО.СписокЛицУправленияОграничен);
	СтДанныхОСАГО.Вставить("ВодителиТСсКБМ", ВодителиТСсКБМ);

	ТзКБМпоТС = Новый ТаблицаЗначений;
	ДобавитьКолонкуВТЗ(ТзКБМпоТС, "Объект");
	ДобавитьКолонкуВТЗ(ТзКБМпоТС, "Класс");
	СформироватьНачТЗПоРегиструПоИзмерению("Объект", ОбъектОСАГО, ТзКБМпоТС, , "КБМпоТС", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен, КонецПериода);

	СтДанныхОСАГО.Вставить("Класс", ТзКБМпоТС[0].Класс);
	СтДанныхОСАГО.Вставить("ИмеютсяНарушения", ДанныеПоОбъектуОСАГО.ИмеютсяНарушения);
	СтДанныхОСАГО.Вставить("ПериодыДляВвода", ПериодыДляВвода);

	Результат	= Новый Структура;
	Результат	= РассчитатьПремиюОСАГО(СтДат, Страхователь, СтДанныхОСАГО, Версия);

	Если ДокументОбъект.Метаданные().Имя="ДопсДСП" Тогда
		
		// получим шапку исходного договора
		СтОтбора		= Новый Структура("Договор", ДокументОбъект.ИсходныйДоговор);
		СтШапки			= РегистрыСведений.ШапкаДСП.ПолучитьПоследнее(Дата-86400, СтОтбора);	
		мНачСрокСтрахования	= ПолучитьРазницуДатВМесяцахНеполных(СтШапки.ДатаНачала, СтШапки.ДатаОкончания);
		
		ТзНачОбъектыИРиски	= РегистрыСведений.ОбъектыИРискиДСП.СрезПоследних(Дата-86400, СтОтбора);
		мНачПремия			= 0;
		Для Каждого СтрНач Из ТзНачОбъектыИРиски Цикл
			мНачПремия	= мНачПремия + СтрНач.Премия;
		КонецЦикла;
		
		ТекПремия	= Результат.Премия;
		Если ТекПремия<>мНачПремия Тогда
			Если ТекПремия<мНачПремия Тогда
				НачПремияСРасходами	= мНачПремия * 0.97;
				ФормулаНачПремия	= "0,97*По";
			Иначе
				НачПремияСРасходами	= мНачПремия;
				ФормулаНачПремия	= "По";
			КонецЕсли;
			ДатаНачалаДоговора		= ДатаНачала;
			ДатаТекущая				= Дата;
			ДатаОкончанияДоговора	= ДатаОкончания;
			СрокТекущий				= ПолучитьРазницуДатВМесяцахНеполных(ДатаНачалаДоговора, ДатаТекущая);
			СрокСтрахования			= ПолучитьРазницуДатВМесяцахНеполных(ДатаНачалаДоговора, ДатаОкончанияДоговора);
			Если СрокСтрахования=мНачСрокСтрахования Тогда
				СтраховаяПремия			= мНачПремия + (ТекПремия-НачПремияСРасходами)*(СрокСтрахования-СрокТекущий)/СрокСтрахования;
				ФормулаДоплата	= "(" + Результат.Формула + " - " + ФормулаНачПремия + ")*(" + Строка(СрокСтрахования)
					+ " - " + Строка(СрокТекущий) + ")/" + Строка(СрокСтрахования);
			Иначе
				ИзмПремия	= ТекПремия*(СрокСтрахования-СрокТекущий)/СрокСтрахования - НачПремияСРасходами*(мНачСрокСтрахования-СрокТекущий)/мНачСрокСтрахования;
				СтраховаяПремия		= мНачПремия + ИзмПремия;
				ФормулаДоплата	= Результат.Формула + "*(" + Строка(СрокСтрахования)
					+ " - " + Строка(СрокТекущий) + ")/" + Строка(СрокСтрахования) + " - "
					+ ФормулаНачПремия + "*(" + Строка(мНачСрокСтрахования)
					+ " - " + Строка(СрокТекущий) + ")/" + Строка(мНачСрокСтрахования);
			КонецЕсли;
			Результат.Премия	= СтраховаяПремия;
			ФормулаПремияДопс	= ФормулаНачПремия + " + " + ФормулаДоплата;
		КонецЕсли;
	КонецЕсли;

	Обл	= Макет.ПолучитьОбласть("Шапка");

	// полис
	ТзОформленныеБСО = Новый ТаблицаЗначений;
	ДобавитьКолонкуВТЗ(ТзОформленныеБСО, "Объект");
	ДобавитьКолонкуВТЗ(ТзОформленныеБСО, "Вид");
	ДобавитьКолонкуВТЗ(ТзОформленныеБСО, "Номер");
	ДобавитьКолонкуВТЗ(ТзОформленныеБСО, "Серия");
	СформироватьНачТЗПоРегистру(ТзОформленныеБСО, Договор, "ОформленныеБСО", ДокументОбъект.Ссылка, мЯвляетсяДопсом, ДокументОбъект.Проведен, КонецПериода);
	Для Каждого СтрокаБСО Из ТзОформленныеБСО Цикл
		Если СтрокаБСО.Вид.Тип=Перечисления.ТипыБСО.ПолисОсаго Тогда
			Прервать;
		КонецЕсли;
	КонецЦикла;
	Если СтрокаБСО=Неопределено Тогда
		Обл.Параметры.ПечПолисСерия	= "";
		Обл.Параметры.ПечПолисНомер	= "";
	Иначе
		Обл.Параметры.ПечПолисСерия=СокрЛП(СтрокаБСО.Серия);
		Обл.Параметры.ПечПолисНомер=СокрЛП(СтрокаБСО.Номер);
	КонецЕсли;
	ТабДок.Вывести(Обл);

	Обл	= Макет.ПолучитьОбласть("ТБ");
	Обл.Параметры.ПечТипТС	= Строка(ТипТСПоОСАГО);
	Обл.Параметры.ПечТБ		= Строка(Результат.ТБ) + " руб.";
	ТабДок.Вывести(Обл);

	Если НЕ ЭтоПрицеп Тогда
		Обл	= Макет.ПолучитьОбласть("КБМ");
		Обл.Параметры.ПечКласс	= Строка(Результат.Класс);
		Обл.Параметры.ПечКБМ	= Строка(Результат.КБМ);
		ТабДок.Вывести(Обл);

		Обл	= Макет.ПолучитьОбласть("КВС");
		Обл.Параметры.ПечВозрастСтаж	= Результат.ВозрастСтаж;
		Обл.Параметры.ПечКВС			= Строка(Результат.КВС);
		ТабДок.Вывести(Обл);

		Обл	= Макет.ПолучитьОбласть("КО");
		Если	Результат.КО=1											И
				Собственник.ЮрФизЛицо=Перечисления.ЮрФизЛицо.ФизЛицо	Тогда
			Обл.Параметры.ПечКолвоЛиц = "Количество лиц, допущенных к управлению ТС, ограничено.";
		Иначе
			Обл.Параметры.ПечКолвоЛиц = "Количество лиц, допущенных к управлению ТС, не ограничено.";
		КонецЕсли;
		Обл.Параметры.ПечКО	= Строка(Результат.КО);
		ТабДок.Вывести(Обл);
		
		Если ЭтоЛегковой Тогда
			Обл	= Макет.ПолучитьОбласть("КМ");
			Обл.Параметры.ПечМощность	= Строка(ТС.МощностьДвигателя);
			Обл.Параметры.ПечКМ			= Строка(Результат.КМ);
			ТабДок.Вывести(Обл);
		КонецЕсли;

		Обл	= Макет.ПолучитьОбласть("КН");
		Если Результат.КН=1.5 Тогда
			Обл.Параметры.ПечНарушения = "Нарушения имеются.";
		Иначе
			Обл.Параметры.ПечНарушения = "Нарушений нет.";
		КонецЕсли;
		Обл.Параметры.ПечКН	= Строка(Результат.КН);
		ТабДок.Вывести(Обл);
	КонецЕсли;

	Обл	= Макет.ПолучитьОбласть("КТ");

	СтОтбора = Новый Структура("Контрагент", Собственник);
	СтНастройки = РегистрыСведений.НастройкиКонтрагентовПоОСАГО.ПолучитьПоследнее(ДатаВерсии, СтОтбора);
	СвойстваПоОсагоОпределены = ?(СтНастройки.Количество() > 0, Истина, Ложь);
	Если СвойстваПоОсагоОпределены Тогда
		ТерриторияОСАГО = СтНастройки.Территория;
	КонецЕсли;

	Обл.Параметры.ПечРегион	= ТерриторияОСАГО.Наименование;
	Обл.Параметры.ПечКТ		= Строка(Результат.КТ);
	ТабДок.Вывести(Обл);

	Обл	= Макет.ПолучитьОбласть("КС");
	Если Результат.КС=0.7 Тогда
		Обл.Параметры.ПечПериодИспользования = "Период использования ТС до 6 месяцев включительно.";
	ИначеЕсли Результат.КС=0.8 Тогда
		Обл.Параметры.ПечПериодИспользования = "Период использования ТС до 7 месяцев включительно.";
	ИначеЕсли Результат.КС=0.9 Тогда
		Обл.Параметры.ПечПериодИспользования = "Период использования ТС до 8 месяцев включительно.";
	ИначеЕсли Результат.КС=0.95 Тогда
		Обл.Параметры.ПечПериодИспользования = "Период использования ТС до 9 месяцев включительно.";
	Иначе
		Обл.Параметры.ПечПериодИспользования = "Период использования ТС свыше 9 месяцев.";
	КонецЕсли;
	Обл.Параметры.ПечКС	= Строка(Результат.КС);
	ТабДок.Вывести(Обл);
	
	Обл	= Макет.ПолучитьОбласть("КП");
	Если Результат.КП=0.2 Тогда
		Если ЗначениеНеЗаполнено(ТС.МестоРегистрации) Тогда
			Обл.Параметры.ПечСрокСтрахования = "Срок страхования ТС до 20 дней включительно.";
		Иначе		
			Обл.Параметры.ПечСрокСтрахования = "Срок страхования ТС до 15 дней включительно.";
		КонецЕсли;
	ИначеЕсли Результат.КП=0.3 Тогда
		Обл.Параметры.ПечСрокСтрахования = "Срок страхования ТС от 16 дней до 1 месяца включительно.";
	ИначеЕсли Результат.КП=0.4 Тогда
		Обл.Параметры.ПечСрокСтрахования = "Срок страхования ТС до 2 месяцев включительно.";
	ИначеЕсли Результат.КП=0.5 Тогда
		Обл.Параметры.ПечСрокСтрахования = "Срок страхования ТС до 3 месяцев включительно.";
